cmake_minimum_required(VERSION 3.20)
project(CarbonLuauPersistenceDurabilityProbe LANGUAGES C CXX)
option(PERSISTENCE_PROBE_SANITIZE "Instrument investigation fixtures and SQLite (Linux only)" OFF)
if(PERSISTENCE_PROBE_SANITIZE)
    if(NOT UNIX OR APPLE)
        message(FATAL_ERROR "Probe sanitizer configuration is Linux-only")
    endif()
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()

# Standalone investigation only. Never linked into, or shipped with, CarbonLuau.
set(SQLITE_AMALGAMATION_DIR "" CACHE PATH "Verified SQLite 3.53.4 amalgamation")
foreach(Source sqlite3.c sqlite3.h)
    if(NOT EXISTS "${SQLITE_AMALGAMATION_DIR}/${Source}")
        message(FATAL_ERROR "Supply the verified SQLite amalgamation directory")
    endif()
endforeach()
file(SHA256 "${SQLITE_AMALGAMATION_DIR}/sqlite3.c" SourceHash)
file(SHA256 "${SQLITE_AMALGAMATION_DIR}/sqlite3.h" HeaderHash)
if(NOT SourceHash STREQUAL "b1dd5d74ec7f29055a6684fa06fb3c2f6821c87dd38f9a458dfd2e8a1db28189"
   OR NOT HeaderHash STREQUAL "919e7f2e8ed1d8f56ac17b412b8971c76aa5d1a879752cc6058f75e7d5910e1d")
    message(FATAL_ERROR "SQLite source differs from the investigated pin")
endif()
add_library(PersistenceProbeSqlite STATIC "${SQLITE_AMALGAMATION_DIR}/sqlite3.c")
target_include_directories(PersistenceProbeSqlite PUBLIC "${SQLITE_AMALGAMATION_DIR}")
target_compile_definitions(PersistenceProbeSqlite PRIVATE SQLITE_THREADSAFE=0 SQLITE_OMIT_LOAD_EXTENSION)
# Keep historical mode-comparison probes WAL-capable. Current Backend fixtures
# use the same no-WAL configuration as the shipped private worker.
add_library(PersistenceBackendSqlite STATIC "${SQLITE_AMALGAMATION_DIR}/sqlite3.c")
target_include_directories(PersistenceBackendSqlite PUBLIC "${SQLITE_AMALGAMATION_DIR}")
target_compile_definitions(PersistenceBackendSqlite PRIVATE SQLITE_THREADSAFE=0 SQLITE_OMIT_LOAD_EXTENSION SQLITE_OMIT_WAL)
add_executable(PersistenceDurabilityProbe DurabilityProbe.cpp)
target_compile_features(PersistenceDurabilityProbe PRIVATE cxx_std_17)
target_link_libraries(PersistenceDurabilityProbe PRIVATE PersistenceProbeSqlite)
if(UNIX AND NOT APPLE)
    target_link_options(PersistenceDurabilityProbe PRIVATE
        -Wl,--wrap=fsync -Wl,--wrap=fdatasync -Wl,--wrap=unlink)
    target_link_libraries(PersistenceDurabilityProbe PRIVATE m)
elseif(NOT WIN32)
    message(FATAL_ERROR "Investigation supports Windows and glibc Linux only")
endif()
enable_testing()
add_test(NAME PersistenceDurabilityObservation COMMAND PersistenceDurabilityProbe)
set_tests_properties(PersistenceDurabilityObservation PROPERTIES TIMEOUT 15)
add_executable(PersistenceModeProbe ModeProbe.cpp)
target_compile_features(PersistenceModeProbe PRIVATE cxx_std_17)
target_link_libraries(PersistenceModeProbe PRIVATE PersistenceProbeSqlite)
if(UNIX)
    target_link_libraries(PersistenceModeProbe PRIVATE m)
endif()
add_test(NAME PersistenceModeInvestigation COMMAND PersistenceModeProbe)
set_tests_properties(PersistenceModeInvestigation PROPERTIES TIMEOUT 180)
add_executable(PersistenceFormatTests FormatTests.cpp ../../native/src/persistence/Format.cpp)
target_include_directories(PersistenceFormatTests PRIVATE ../../native/src/persistence)
target_compile_features(PersistenceFormatTests PRIVATE cxx_std_17)
add_test(NAME PersistenceFormat COMMAND PersistenceFormatTests)
set_tests_properties(PersistenceFormat PROPERTIES TIMEOUT 15)
add_executable(PersistenceBackendTests BackendTests.cpp ../../native/src/persistence/Backend.cpp ../../native/src/persistence/Format.cpp)
target_include_directories(PersistenceBackendTests PRIVATE ../../native/src/persistence)
target_compile_features(PersistenceBackendTests PRIVATE cxx_std_17)
target_link_libraries(PersistenceBackendTests PRIVATE PersistenceBackendSqlite)
if(UNIX)
    target_link_libraries(PersistenceBackendTests PRIVATE m)
endif()
add_test(NAME PersistenceBackend COMMAND PersistenceBackendTests)
set_tests_properties(PersistenceBackend PROPERTIES TIMEOUT 60)
add_executable(PersistenceBackendCrashTests BackendCrashTests.cpp ../../native/src/persistence/Backend.cpp ../../native/src/persistence/Format.cpp)
target_include_directories(PersistenceBackendCrashTests PRIVATE ../../native/src/persistence)
target_compile_features(PersistenceBackendCrashTests PRIVATE cxx_std_17)
target_compile_definitions(PersistenceBackendCrashTests PRIVATE CARBONLUAU_PERSISTENCE_TESTING)
target_link_libraries(PersistenceBackendCrashTests PRIVATE PersistenceBackendSqlite)
if(UNIX)
    target_link_libraries(PersistenceBackendCrashTests PRIVATE m)
endif()
add_test(NAME PersistenceBackendCrash COMMAND PersistenceBackendCrashTests)
set_tests_properties(PersistenceBackendCrash PROPERTIES TIMEOUT 180)
add_executable(PersistenceQuotaTests QuotaTests.cpp ../../native/src/persistence/Backend.cpp ../../native/src/persistence/Format.cpp)
target_include_directories(PersistenceQuotaTests PRIVATE ../../native/src/persistence)
target_compile_features(PersistenceQuotaTests PRIVATE cxx_std_17)
target_link_libraries(PersistenceQuotaTests PRIVATE PersistenceBackendSqlite)
if(UNIX)
    target_link_libraries(PersistenceQuotaTests PRIVATE m)
endif()
add_test(NAME PersistenceQuota COMMAND PersistenceQuotaTests)
set_tests_properties(PersistenceQuota PROPERTIES TIMEOUT 600)
add_executable(carbonluau_storage ../../native/src/persistence/Worker.cpp ../../native/src/persistence/Backend.cpp ../../native/src/persistence/Format.cpp)
target_include_directories(carbonluau_storage PRIVATE ../../native/src/persistence)
target_compile_features(carbonluau_storage PRIVATE cxx_std_17)
target_link_libraries(carbonluau_storage PRIVATE PersistenceBackendSqlite)
if(UNIX)
    target_link_libraries(carbonluau_storage PRIVATE m)
endif()
foreach(Fixture LostAck Hang)
    add_executable(StorageFixture${Fixture} ../../native/src/persistence/Worker.cpp ../../native/src/persistence/Backend.cpp ../../native/src/persistence/Format.cpp)
    target_include_directories(StorageFixture${Fixture} PRIVATE ../../native/src/persistence)
    target_compile_features(StorageFixture${Fixture} PRIVATE cxx_std_17)
    target_link_libraries(StorageFixture${Fixture} PRIVATE PersistenceBackendSqlite)
    if(UNIX)
        target_link_libraries(StorageFixture${Fixture} PRIVATE m)
    endif()
endforeach()
target_compile_definitions(StorageFixtureLostAck PRIVATE CARBONLUAU_STORAGE_FIXTURE=1)
target_compile_definitions(StorageFixtureHang PRIVATE CARBONLUAU_STORAGE_FIXTURE=2)
