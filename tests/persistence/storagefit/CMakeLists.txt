cmake_minimum_required(VERSION 3.20)
project(CarbonLuauStorageFitResearch LANGUAGES C CXX)
# Standalone prototype only. Not included by any production/package build.
set(SQLITE_AMALGAMATION_DIR "" CACHE PATH "Exact qualified SQLite sources")
file(SHA256 "${SQLITE_AMALGAMATION_DIR}/sqlite3.c" SourceHash)
file(SHA256 "${SQLITE_AMALGAMATION_DIR}/sqlite3.h" HeaderHash)
if(NOT SourceHash STREQUAL "b1dd5d74ec7f29055a6684fa06fb3c2f6821c87dd38f9a458dfd2e8a1db28189" OR
   NOT HeaderHash STREQUAL "919e7f2e8ed1d8f56ac17b412b8971c76aa5d1a879752cc6058f75e7d5910e1d")
  message(FATAL_ERROR "Unqualified SQLite pin")
endif()
add_library(StorageFitSqlite STATIC "${SQLITE_AMALGAMATION_DIR}/sqlite3.c")
target_include_directories(StorageFitSqlite PUBLIC "${SQLITE_AMALGAMATION_DIR}")
target_compile_definitions(StorageFitSqlite PRIVATE SQLITE_THREADSAFE=0 SQLITE_OMIT_LOAD_EXTENSION SQLITE_OMIT_WAL SQLITE_ENABLE_DBSTAT_VTAB)
add_executable(StorageFitProbe Probe.cpp ../../../native/src/persistence/Format.cpp ../../../native/src/persistence/Derived.cpp ../../../native/src/persistence/Sortable.cpp)
target_compile_features(StorageFitProbe PRIVATE cxx_std_17)
target_include_directories(StorageFitProbe PRIVATE ../../../native/src/persistence)
target_link_libraries(StorageFitProbe PRIVATE StorageFitSqlite)
if(UNIX)
  target_link_libraries(StorageFitProbe PRIVATE m)
endif()
add_executable(StorageMigrationProbe Migration.cpp ../../../native/src/persistence/Format.cpp ../../../native/src/persistence/Derived.cpp ../../../native/src/persistence/Sortable.cpp)
target_compile_features(StorageMigrationProbe PRIVATE cxx_std_17)
target_include_directories(StorageMigrationProbe PRIVATE ../../../native/src/persistence)
target_link_libraries(StorageMigrationProbe PRIVATE StorageFitSqlite)
if(UNIX)
  target_link_libraries(StorageMigrationProbe PRIVATE m)
endif()
add_executable(StorageShapeProbe Shapes.cpp ../../../native/src/persistence/Format.cpp ../../../native/src/persistence/Derived.cpp ../../../native/src/persistence/Sortable.cpp)
target_compile_features(StorageShapeProbe PRIVATE cxx_std_17)
target_include_directories(StorageShapeProbe PRIVATE ../../../native/src/persistence)
target_link_libraries(StorageShapeProbe PRIVATE StorageFitSqlite)
if(UNIX)
  target_link_libraries(StorageShapeProbe PRIVATE m)
