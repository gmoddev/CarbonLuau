cmake_minimum_required(VERSION 3.20)
project(CarbonLuauAllocationInvestigation LANGUAGES C CXX)
# Standalone research targets only; no production build/package change.
add_subdirectory(.. persistence)
add_executable(PersistenceAllocationStartupProbe ../AllocationStartupProbe.cpp
    ../../../native/src/persistence/Backend.cpp ../../../native/src/persistence/Format.cpp)
target_compile_features(PersistenceAllocationStartupProbe PRIVATE cxx_std_17)
target_include_directories(PersistenceAllocationStartupProbe PRIVATE ../../../native/src/persistence)
target_link_libraries(PersistenceAllocationStartupProbe PRIVATE PersistenceBackendSqlite)
if(UNIX)
    target_link_libraries(PersistenceAllocationStartupProbe PRIVATE m)
endif()
if(PERSISTENCE_PROBE_SANITIZE)
    target_compile_options(PersistenceAllocationStartupProbe PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(PersistenceAllocationStartupProbe PRIVATE -fsanitize=address,undefined)
endif()
