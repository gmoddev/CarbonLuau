cmake_minimum_required(VERSION 3.20)
project(CarbonLuauAnalysis LANGUAGES CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(LUAU_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(LUAU_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LUAU_BUILD_WEB OFF CACHE BOOL "" FORCE)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    set(LUAU_STATIC_CRT ON CACHE BOOL "" FORCE)
endif()
add_subdirectory(../third_party/luau luau EXCLUDE_FROM_ALL)
if(MSVC)
    target_compile_options(Luau.Ast PRIVATE /MP1)
    target_compile_options(Luau.Common PRIVATE /MP1)
endif()
add_library(carbonluau_analysis SHARED SourceAnalysis.cpp)
target_compile_features(carbonluau_analysis PRIVATE cxx_std_17)
target_link_libraries(carbonluau_analysis PRIVATE Luau.Ast)
add_executable(carbonluau-analysis-launcher AnalysisLauncher.cpp)
target_compile_features(carbonluau-analysis-launcher PRIVATE cxx_std_17)
option(CARBONLUAU_ANALYSIS_TESTS "Build analysis containment fixtures" OFF)
if(CARBONLUAU_ANALYSIS_TESTS)
    add_executable(carbonluau-analysis-fixture AnalysisFixture.cpp)
    target_compile_features(carbonluau-analysis-fixture PRIVATE cxx_std_17)
endif()
# Deliberately no VM, Carbon, Rust, Unity or production native library dependency.
