cmake_minimum_required(VERSION 3.20)
project(CarbonLuauNative VERSION 0.0.1 LANGUAGES CXX)

if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64)$")
    message(FATAL_ERROR "Phase 0 supports x64 only")
endif()
if(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Phase 0 supports Windows and Linux only")
endif()

add_library(carbonluau_native SHARED src/probe.cpp)
target_include_directories(carbonluau_native PRIVATE include)
set_target_properties(carbonluau_native PROPERTIES CXX_VISIBILITY_PRESET hidden POSITION_INDEPENDENT_CODE ON)
include(CTest)
if(BUILD_TESTING)
    add_executable(ProbeHost ../tests/native/ProbeHost.cpp)
    target_link_libraries(ProbeHost PRIVATE ${CMAKE_DL_LIBS})
    add_test(NAME NativeLoadUnload COMMAND ProbeHost $<TARGET_FILE:carbonluau_native>)
    add_library(WrongProbe SHARED ../tests/native/WrongProbe.cpp)
    target_include_directories(WrongProbe PRIVATE include)
    add_library(MissingSymbol SHARED ../tests/native/MissingSymbol.cpp)
    set_target_properties(MissingSymbol PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
endif()
