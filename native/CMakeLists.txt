cmake_minimum_required(VERSION 3.20)
project(CarbonLuauNative VERSION 0.4.0 LANGUAGES CXX)

# Rust/Unity can preload an older app-local MSVCP140 than our compiler requires.
# Keep C++ runtime allocation/exceptions private to this DLL and pinned Luau.
# The C ABI never transfers CRT-owned objects or ownership to the caller.
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    set(LUAU_STATIC_CRT ON CACHE BOOL "" FORCE)
endif()

option(CARBONLUAU_SANITIZE "Instrument runtime and pinned dependencies with ASan/UBSan" OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
if(CARBONLUAU_SANITIZE)
    if(MSVC)
        message(FATAL_ERROR "Combined sanitizers require the Linux toolchain")
    endif()
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
set(LUAU_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(LUAU_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LUAU_BUILD_WEB OFF CACHE BOOL "" FORCE)
set(LUAU_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(LUAU_EXTERN_C OFF CACHE BOOL "Use upstream C++ protected errors, not longjmp" FORCE)
add_subdirectory(third_party/luau EXCLUDE_FROM_ALL)
# The vendor enables unbounded /MP. One compile per MSBuild project keeps
# --parallel 4 bounded without altering the pinned dependency.
if(MSVC)
    foreach(Target Luau.Common Luau.Ast Luau.Bytecode Luau.Compiler Luau.VM)
        target_compile_options(${Target} PRIVATE /MP1)
    endforeach()
endif()

if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64)$")
    message(FATAL_ERROR "Phase 0 supports x64 only")
endif()
if(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Phase 0 supports Windows and Linux only")
endif()

set(CARBONLUAU_RUNTIME_SOURCES
    src/runtime/VmRegistry.cpp
    src/runtime/VmState.cpp
    src/runtime/Publication.cpp
    src/runtime/Deadline.cpp
    src/scripts/Compiler.cpp
    src/scripts/ModuleLoader.cpp
    src/facade/FacadeBridge.cpp)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/third_party/LUAU_REVISION.txt" LUAU_REVISION)
string(REGEX MATCH "Pinned commit: ([0-9a-f]+)" LUAU_PIN_MATCH "${LUAU_REVISION}")
set(LUAU_REVISION "${CMAKE_MATCH_1}")
string(LENGTH "${LUAU_REVISION}" LUAU_REVISION_LENGTH)
if(NOT LUAU_REVISION_LENGTH EQUAL 40)
    message(FATAL_ERROR "Expected a full pinned Luau revision")
endif()
if(CARBONLUAU_SANITIZE)
    add_compile_definitions(CARBONLUAU_SANITIZE)
endif()
add_executable(carbonluau_compiler src/scripts/CompilerWorker.cpp)
target_compile_features(carbonluau_compiler PRIVATE cxx_std_17)
target_compile_definitions(carbonluau_compiler PRIVATE CARBONLUAU_REVISION="${LUAU_REVISION}")
target_link_libraries(carbonluau_compiler PRIVATE Luau.Compiler)
set_target_properties(carbonluau_compiler PROPERTIES OUTPUT_NAME "carbonluau_compiler")
add_library(carbonluau_native SHARED src/probe.cpp src/Runtime.cpp ${CARBONLUAU_RUNTIME_SOURCES})
add_dependencies(carbonluau_native carbonluau_compiler)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../scripts/bootstrap.luau" CARBONLUAU_BOOTSTRAP)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../scripts/bootstrap.luau")
string(LENGTH "${CARBONLUAU_BOOTSTRAP}" CARBONLUAU_BOOTSTRAP_REMAINING)
set(CARBONLUAU_BOOTSTRAP_OFFSET 0)
set(CARBONLUAU_BOOTSTRAP_LITERALS "")
while(CARBONLUAU_BOOTSTRAP_REMAINING GREATER 0)
    if(CARBONLUAU_BOOTSTRAP_REMAINING GREATER 8000)
        set(CARBONLUAU_BOOTSTRAP_CHUNK_LENGTH 8000)
    else()
        set(CARBONLUAU_BOOTSTRAP_CHUNK_LENGTH ${CARBONLUAU_BOOTSTRAP_REMAINING})
    endif()
    string(SUBSTRING "${CARBONLUAU_BOOTSTRAP}" ${CARBONLUAU_BOOTSTRAP_OFFSET} ${CARBONLUAU_BOOTSTRAP_CHUNK_LENGTH} CARBONLUAU_BOOTSTRAP_CHUNK)
    string(APPEND CARBONLUAU_BOOTSTRAP_LITERALS "R\"CLBOOT(${CARBONLUAU_BOOTSTRAP_CHUNK})CLBOOT\"\n")
    math(EXPR CARBONLUAU_BOOTSTRAP_OFFSET "${CARBONLUAU_BOOTSTRAP_OFFSET} + ${CARBONLUAU_BOOTSTRAP_CHUNK_LENGTH}")
    math(EXPR CARBONLUAU_BOOTSTRAP_REMAINING "${CARBONLUAU_BOOTSTRAP_REMAINING} - ${CARBONLUAU_BOOTSTRAP_CHUNK_LENGTH}")
endwhile()
configure_file(src/Bootstrap.h.in "${CMAKE_CURRENT_BINARY_DIR}/generated/Bootstrap.h" @ONLY)
include_directories("${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_features(carbonluau_native PRIVATE cxx_std_17)
target_compile_definitions(carbonluau_native PRIVATE CARBONLUAU_REVISION="${LUAU_REVISION}")
target_link_libraries(carbonluau_native PRIVATE Luau.VM ${CMAKE_DL_LIBS})
target_include_directories(carbonluau_native PRIVATE include)
set_target_properties(carbonluau_native PROPERTIES CXX_VISIBILITY_PRESET hidden POSITION_INDEPENDENT_CODE ON)
include(CTest)
if(BUILD_TESTING)
    add_executable(ScriptTests ../tests/native/ScriptTests.cpp)
    target_include_directories(ScriptTests PRIVATE include)
    target_link_libraries(ScriptTests PRIVATE carbonluau_native)
    add_test(NAME ScriptCore COMMAND ScriptTests)
    set_tests_properties(ScriptCore PROPERTIES TIMEOUT 180)
    add_executable(RuntimeFaultTests ../tests/native/RuntimeFaultTests.cpp src/Runtime.cpp ${CARBONLUAU_RUNTIME_SOURCES})
    target_include_directories(RuntimeFaultTests PRIVATE include "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_compile_definitions(RuntimeFaultTests PRIVATE CARBONLUAU_REVISION="${LUAU_REVISION}" CARBONLUAU_TESTING)
    target_link_libraries(RuntimeFaultTests PRIVATE Luau.VM ${CMAKE_DL_LIBS})
    foreach(Fixture Hang Crash Malformed Truncated Oversized Revision Nonce Memory)
        add_executable(CompilerFixture${Fixture} ../tests/native/CompilerFixture.cpp)
        target_compile_features(CompilerFixture${Fixture} PRIVATE cxx_std_17)
        target_compile_definitions(CompilerFixture${Fixture} PRIVATE CARBONLUAU_REVISION="${LUAU_REVISION}")
    endforeach()
    target_compile_definitions(CompilerFixtureHang PRIVATE CARBONLUAU_COMPILER_FIXTURE=1)
    target_compile_definitions(CompilerFixtureCrash PRIVATE CARBONLUAU_COMPILER_FIXTURE=2)
    target_compile_definitions(CompilerFixtureMalformed PRIVATE CARBONLUAU_COMPILER_FIXTURE=3)
    target_compile_definitions(CompilerFixtureTruncated PRIVATE CARBONLUAU_COMPILER_FIXTURE=4)
    target_compile_definitions(CompilerFixtureOversized PRIVATE CARBONLUAU_COMPILER_FIXTURE=5)
    target_compile_definitions(CompilerFixtureRevision PRIVATE CARBONLUAU_COMPILER_FIXTURE=6)
    target_compile_definitions(CompilerFixtureNonce PRIVATE CARBONLUAU_COMPILER_FIXTURE=7)
    target_compile_definitions(CompilerFixtureMemory PRIVATE CARBONLUAU_COMPILER_FIXTURE=8)
    add_dependencies(RuntimeFaultTests carbonluau_compiler CompilerFixtureHang CompilerFixtureCrash)
    add_test(NAME RuntimeAllocationFaults COMMAND RuntimeFaultTests $<TARGET_FILE:carbonluau_compiler>
        $<TARGET_FILE:CompilerFixtureHang> $<TARGET_FILE:CompilerFixtureCrash>)
    set_tests_properties(RuntimeAllocationFaults PROPERTIES TIMEOUT 180)
    add_executable(CompilerContainmentTests ../tests/native/CompilerContainmentTests.cpp src/scripts/Compiler.cpp)
    target_compile_features(CompilerContainmentTests PRIVATE cxx_std_17)
    target_compile_definitions(CompilerContainmentTests PRIVATE CARBONLUAU_REVISION="${LUAU_REVISION}" CARBONLUAU_TESTING)
    target_link_libraries(CompilerContainmentTests PRIVATE Luau.Compiler ${CMAKE_DL_LIBS})
    add_dependencies(CompilerContainmentTests carbonluau_compiler CompilerFixtureHang CompilerFixtureCrash
        CompilerFixtureMalformed CompilerFixtureTruncated CompilerFixtureOversized CompilerFixtureRevision CompilerFixtureNonce)
    add_dependencies(CompilerContainmentTests CompilerFixtureMemory)
    add_test(NAME CompilerContainment COMMAND CompilerContainmentTests
        $<TARGET_FILE:carbonluau_compiler> $<TARGET_FILE:CompilerFixtureHang> $<TARGET_FILE:CompilerFixtureCrash>
        $<TARGET_FILE:CompilerFixtureMalformed> $<TARGET_FILE:CompilerFixtureTruncated> $<TARGET_FILE:CompilerFixtureOversized>
        $<TARGET_FILE:CompilerFixtureRevision> $<TARGET_FILE:CompilerFixtureNonce> $<TARGET_FILE:CompilerFixtureMemory>)
    set_tests_properties(CompilerContainment PROPERTIES TIMEOUT 30)
    add_library(WrongAbi SHARED ../tests/native/WrongAbi.cpp)
    add_library(LegacyProbe SHARED src/probe.cpp)
    target_include_directories(WrongAbi PRIVATE include)
    target_include_directories(LegacyProbe PRIVATE include)
    add_executable(RuntimeTests ../tests/native/RuntimeTests.cpp)
    target_include_directories(RuntimeTests PRIVATE include)
    target_link_libraries(RuntimeTests PRIVATE carbonluau_native)
    add_test(NAME RuntimeCore COMMAND RuntimeTests)
    set_tests_properties(RuntimeCore PROPERTIES TIMEOUT 120)
    add_executable(ProbeHost ../tests/native/ProbeHost.cpp)
    target_link_libraries(ProbeHost PRIVATE ${CMAKE_DL_LIBS})
    add_test(NAME NativeLoadUnload COMMAND ProbeHost $<TARGET_FILE:carbonluau_native>)
    add_library(WrongProbe SHARED ../tests/native/WrongProbe.cpp)
    target_include_directories(WrongProbe PRIVATE include)
    add_library(MissingSymbol SHARED ../tests/native/MissingSymbol.cpp)
    set_target_properties(MissingSymbol PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
endif()
